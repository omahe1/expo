const { ethers } = require('ethers');
require('dotenv').config(); // Load the variables from .env file

// --- 1. CONFIGURATION ---
const PRIVATE_KEY = process.env.PRIVATE_KEY; 
const RPC_URL = process.env.RPC_URL;
const OMCOIN_CONTRACT_ADDRESS = process.env.OMCOIN_CONTRACT_ADDRESS;

// The ABI snippet for the 'transfer' function (the minimum required)
const OMCOIN_ABI = [
    // Function to transfer tokens from one address to another
    "function transfer(address to, uint256 amount) public returns (bool)" 
];

// --- 2. CORE TRANSFER FUNCTION ---
async function rewardUserWithOMCOIN(recipientAddress, amountToReward) {
    try {
        // 2a. Connect to the BSC network
        const provider = new ethers.providers.JsonRpcProvider(RPC_URL);

        // 2b. Create a wallet instance for signing (This uses the server's private key)
        const signer = new ethers.Wallet(PRIVATE_KEY, provider);

        // 2c. Connect to the OMCOIN contract using its address, ABI, and the signer
        const omcoinContract = new ethers.Contract(
            OMCOIN_CONTRACT_ADDRESS,
            OMCOIN_ABI,
            signer
        );

        // 2d. Format the amount using the token's decimals (18 in your case)
        // This converts 10 OMCOIN into the massive number the smart contract expects (10 * 10^18)
        const amountInWei = ethers.utils.parseUnits(amountToReward.toString(), 18);

        // 2e. Execute the transfer transaction!
        console.log(`Sending ${amountToReward} OMCOIN to ${recipientAddress}...`);

        const tx = await omcoinContract.transfer(
            recipientAddress, 
            amountInWei,
            // You can optionally specify gas limit/price for more control
        );

        console.log('Transaction hash:', tx.hash);

        // Wait for the transaction to be mined (confirmed on the blockchain)
        const receipt = await tx.wait();
        console.log('Transaction confirmed in block:', receipt.blockNumber);

        return { success: true, hash: tx.hash };

    } catch (error) {
        console.error("OMCOIN Reward Failed:", error.message);
        return { success: false, error: error.message };
    }
}
